# --var sets a variable (it wins over DOZE_VAR_*), and -c names a folder whose *.hcl files are read together (Go's -c
# <dir>): status with the daemon down shows what they declare.
@ project bare
@ env DOZE_VAR_port=7000
@ file project/split/a.hcl variable "port" { default = 6000 }\nstack = "split"\n
@ file project/split/b.hcl app "web" {\n  run  = "serve"\n  port = var.port\n}\n
